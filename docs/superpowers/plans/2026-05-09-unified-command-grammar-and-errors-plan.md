# Unified Command Grammar + Structured Errors Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Move command grammar (slash + CLI + HTTP + URI) and error vocabulary into a single source per command module via a `command_meta/0` callback. Generate `slash-routes.default.yaml` + `docs/grammar/{commands,errors}.md` from that source. CI gate prevents drift. Add bare-prefix help routing as a UX byproduct.

**Architecture:** A thin DSL (`use Esr.Commands.Meta` + `command :kind do ... end` block) compiles into a `%Esr.Commands.Meta.Spec{}` struct returned by `command_meta/0`. A single `Esr.Commands.Render` module turns specs into help lines, module help, and error maps. Two mix tasks (`gen_slash_routes`, `gen_command_docs`) walk every command module and emit the yaml + docs files; one CI gate (`check_command_docs`) regenerates and diffs. The slash router learns three new shapes: `/<resource>`, `/<resource>:help`, `/<unknown>`.

**Tech Stack:** Elixir 1.19 / Phoenix 1.8 / `yaml_elixir` (parse) + `Ymlr` (emit, added Phase 1) / ExUnit / GitHub Actions.

```elixir
defmodule Esr.Commands.Meta do
  @moduledoc """
  DSL + behaviour for declaring a command's grammar (slash + CLI shape,
  args, errors, help text) in one place. The DSL block expands at compile
  time into a `command_meta/0` function returning a `%Spec{}` struct.

  Consumers:
    * `Esr.Commands.Render` — turns specs into help lines / module help /
      error maps.
    * `Mix.Tasks.Esr.GenSlashRoutes` — emits slash-routes.default.yaml.
    * `Mix.Tasks.Esr.GenCommandDocs` — emits docs/grammar/{commands,errors}.md.
    * `Mix.Tasks.Esr.CheckCommandDocs` — CI drift gate.

  Phase 1 (this commit) ships the structs + behaviour declaration. The
  DSL macros land in the next commit; the bulk module conversions happen
  in Phase 3.
  """

  @callback command_meta() :: Esr.Commands.Meta.Spec.t()

  defmodule Spec do
    @moduledoc "Compiled command grammar; the return shape of command_meta/0."

    @type t :: %__MODULE__{
            kind: atom(),
            slash: String.t() | :none,
            category: String.t(),
            description: String.t(),
            permission: String.t() | nil,
            requires_user_binding: boolean(),
            requires_workspace_binding: boolean(),
            args: [Esr.Commands.Meta.Arg.t()],
            errors: [Esr.Commands.Meta.ErrorDecl.t()]
          }

    defstruct kind: nil,
              slash: nil,
              category: "其他",
              description: "",
              permission: nil,
              requires_user_binding: false,
              requires_workspace_binding: false,
              args: [],
              errors: []
  end

  defmodule Arg do
    @moduledoc "Single command arg declaration."

    @type t :: %__MODULE__{
            name: atom(),
            required: boolean(),
            default: term(),
            doc: String.t() | nil
          }

    defstruct name: nil, required: false, default: nil, doc: nil
  end

  defmodule ErrorDecl do
    @moduledoc "Single error code + canonical message."

    @type t :: %__MODULE__{
            code: atom(),
            message: String.t() | nil
          }

    defstruct code: nil, message: nil
  end
end
```

```elixir
  @doc false
  defmacro __using__(_opts) do
    quote do
      import Esr.Commands.Meta, only: [command: 2]
      @behaviour Esr.Commands.Meta
    end
  end

  defmacro command(kind, do: block) when is_atom(kind) do
    quote do
      Module.register_attribute(__MODULE__, :__cmd_args__, accumulate: true)
      Module.register_attribute(__MODULE__, :__cmd_errors__, accumulate: true)
      @__cmd_kind unquote(kind)
      @__cmd_slash nil
      @__cmd_category "其他"
      @__cmd_description ""
      @__cmd_permission nil
      @__cmd_requires_user false
      @__cmd_requires_ws false

      import Esr.Commands.Meta,
        only: [
          slash: 1,
          category: 1,
          description: 1,
          permission: 1,
          requires_user_binding: 1,
          requires_workspace_binding: 1,
          arg: 2,
          error: 1,
          error: 2
        ]

      unquote(block)

      # Compile-time duplicate-error check.
      Esr.Commands.Meta.__validate_no_duplicate_errors__!(
        __MODULE__,
        @__cmd_errors__
      )

      def command_meta do
        %Esr.Commands.Meta.Spec{
          kind: @__cmd_kind,
          slash: @__cmd_slash,
          category: @__cmd_category,
          description: @__cmd_description,
          permission: @__cmd_permission,
          requires_user_binding: @__cmd_requires_user,
          requires_workspace_binding: @__cmd_requires_ws,
          args: Enum.reverse(@__cmd_args__),
          errors: Enum.reverse(@__cmd_errors__)
        }
      end
    end
  end

  defmacro slash(value),       do: quote(do: @__cmd_slash unquote(value))
  defmacro category(value),    do: quote(do: @__cmd_category unquote(value))
  defmacro description(value), do: quote(do: @__cmd_description unquote(value))
  defmacro permission(value),  do: quote(do: @__cmd_permission unquote(value))
  defmacro requires_user_binding(value),      do: quote(do: @__cmd_requires_user unquote(value))
  defmacro requires_workspace_binding(value), do: quote(do: @__cmd_requires_ws unquote(value))

  defmacro arg(name, opts) when is_atom(name) and is_list(opts) do
    quote do
      @__cmd_args__ %Esr.Commands.Meta.Arg{
        name: unquote(name),
        required: Keyword.get(unquote(opts), :required, false),
        default: Keyword.get(unquote(opts), :default, nil),
        doc: Keyword.get(unquote(opts), :doc, nil)
      }
    end
  end

  defmacro error(code) when is_atom(code) do
    quote do
      @__cmd_errors__ %Esr.Commands.Meta.ErrorDecl{code: unquote(code), message: nil}
    end
  end

  defmacro error(code, message) when is_atom(code) and is_binary(message) do
    quote do
      @__cmd_errors__ %Esr.Commands.Meta.ErrorDecl{
        code: unquote(code),
        message: unquote(message)
      }
    end
  end

  @doc false
  def __validate_no_duplicate_errors__!(module, errors) do
    codes = Enum.map(errors, & &1.code)

    case codes -- Enum.uniq(codes) do
      [] ->
        :ok

      dups ->
        raise CompileError,
          file: to_string(module),
          line: 1,
          description: "duplicate error code #{inspect(hd(dups))} in #{inspect(module)}"
    end
  end
```

```elixir
defmodule Esr.Commands.Render do
  @moduledoc """
  Single renderer for command grammar. Consumes `%Esr.Commands.Meta.Spec{}`
  values and produces:

    * `help_line/1` — one line for `/help` (slash + args + short description)
    * `module_help/1` — multi-line listing for `/<resource>` and `/<resource>:help`
    * `error/2` and `error/3` — structured `%{"type" => ..., "message" => ...}` maps
      with the canonical message from the spec, optionally interpolating
      `detail` keys via `%{key}` syntax.

  Phase 1 — Step 3/3 of DSL infrastructure. Phase 3 deletes every
  hand-written error literal in command modules and routes them through
  here.
  """

  alias Esr.Commands.Meta.{Spec, Arg, ErrorDecl}

  @spec help_line(Spec.t()) :: String.t()
  def help_line(%Spec{slash: :none}), do: ""

  def help_line(%Spec{slash: slash, args: args, description: desc}) do
    args_str = render_args(args)
    arg_seg = if args_str == "", do: "", else: " " <> args_str
    "  #{slash}#{arg_seg}\n                   — #{desc}"
  end

  @spec module_help([Spec.t()]) :: String.t()
  def module_help(specs) when is_list(specs) do
    specs
    |> Enum.reject(&match?(%Spec{slash: :none}, &1))
    |> Enum.sort_by(& &1.kind)
    |> Enum.map_join("\n", &help_line/1)
  end

  @spec error(Spec.t(), atom()) :: {:error, map()}
  def error(spec, code), do: error(spec, code, %{})

  @spec error(Spec.t(), atom(), map()) :: {:error, map()}
  def error(%Spec{errors: errors, kind: kind}, code, detail) when is_atom(code) do
    case Enum.find(errors, &(&1.code == code)) do
      %ErrorDecl{message: nil} ->
        {:error, %{"type" => Atom.to_string(code), "message" => generic_message(kind, code)}}

      %ErrorDecl{message: msg} ->
        {:error, %{"type" => Atom.to_string(code), "message" => interpolate(msg, detail)}}

      nil ->
        raise ArgumentError,
              "undeclared error code #{inspect(code)} for command #{inspect(kind)}; " <>
                "add `error #{inspect(code)}, \"...\"` to its command_meta block"
    end
  end

  defp render_args(args) do
    args
    |> Enum.map(fn
      %Arg{name: n, required: true} -> "#{n}=<…>"
      %Arg{name: n} -> "[#{n}=<…>]"
    end)
    |> Enum.join(" ")
  end

  defp generic_message(kind, code) do
    "command #{kind} returned #{code} (no canonical message declared)"
  end

  defp interpolate(template, detail) when is_map(detail) do
    Regex.replace(~r/%\{(\w+)\}/, template, fn _, key ->
      Map.get(detail, String.to_existing_atom(key), "%{#{key}}")
    end)
  end
end
```

```elixir
defmodule Esr.Commands.ResourceHelp do
  @moduledoc """
  Handles bare-prefix help: `/workspace`, `/<resource>:help`, and the
  `/` (no resource) fallback. Pulls route metadata from
  `Esr.Resource.SlashRoute.Registry.list_slashes/0` and renders via
  `Esr.Commands.Render.module_help/1`.

  Wired by `Esr.Entity.SlashHandler` (Phase 6) when input matches a
  bare resource prefix.
  """

  @behaviour Esr.Role.Control

  alias Esr.Resource.SlashRoute.Registry

  @impl true
  @spec execute(map()) :: {:ok, %{required(String.t()) => String.t()}}
  def execute(%{"args" => %{"resource" => res}}) when is_binary(res) and res != "" do
    routes = list_routes_for_resource(res)

    text =
      case routes do
        [] -> unknown_resource_text(res)
        rs -> render_resource(res, rs)
      end

    {:ok, %{"text" => text}}
  end

  def execute(_), do: {:ok, %{"text" => render_top_level()}}

  defp list_routes_for_resource(res) do
    Registry.list_slashes()
    |> Enum.filter(fn route ->
      slash = route[:slash] || ""
      String.starts_with?(slash, "/" <> res <> ":")
    end)
    |> Enum.sort_by(& &1.kind)
  end

  defp render_resource(res, routes) do
    body =
      routes
      |> Enum.map_join("\n", fn route ->
        args_seg = render_args(route.args || [])
        arg_str = if args_seg == "", do: "", else: " " <> args_seg
        "  #{route.slash}#{arg_str}\n                   — #{route.description}"
      end)

    """
    📖 /#{res} — methods

    #{body}

    输入 /<command>:<method> [args...] 调用具体命令；/help 看完整清单。
    """
  end

  defp unknown_resource_text(res) do
    """
    ❌ 未知 resource: `#{res}`

    可用的 top-level resources:

    #{render_top_level_list()}

    输入 /help 看完整命令清单。
    """
  end

  defp render_top_level do
    """
    📖 ESR top-level resources

    #{render_top_level_list()}

    输入 /<resource> 或 /<resource>:help 看该 resource 的方法。
    /help 看完整命令清单。
    """
  end

  defp render_top_level_list do
    Registry.list_slashes()
    |> Enum.map(fn route -> route[:slash] || "" end)
    |> Enum.flat_map(fn
      "/" <> rest ->
        case String.split(rest, ":", parts: 2) do
          [resource, _method] -> [resource]
          _ -> []
        end

      _ ->
        []
    end)
    |> Enum.uniq()
    |> Enum.sort()
    |> Enum.map_join("\n", fn r -> "  /#{r}" end)
  end

  defp render_args(args) do
    args
    |> Enum.map(fn
      %{name: n, required: true} -> "#{n}=<…>"
      %{name: n} -> "[#{n}=<…>]"
    end)
    |> Enum.join(" ")
  end
end
```

```elixir
defmodule Mix.Tasks.Esr.GenSlashRoutes do
  @moduledoc """
  Regenerate `runtime/priv/slash-routes.default.yaml` from every
  Esr.Commands.* module's `command_meta/0` callback. The yaml is
  derived state; this task is the only writer. CI gate
  `mix esr.check_command_docs` diffs in-memory output against the
  on-disk file.

  Usage: `mix esr.gen_slash_routes`
  """

  use Mix.Task
  alias Esr.Commands.Meta.{Spec, Arg}

  @yaml_path "priv/slash-routes.default.yaml"

  @banner """
  # AUTOGENERATED — DO NOT EDIT BY HAND.
  # Source: command_meta/0 in each Esr.Commands.* module.
  # Regenerate: mix esr.gen_slash_routes
  # Drift in this file fails CI via mix esr.check_command_docs.
  """

  @shortdoc "Emit slash-routes.default.yaml from command_meta/0"

  def run(_args) do
    # Load the app so all command modules are compiled + queryable.
    Mix.Task.run("compile")
    Application.ensure_all_started(:esr)

    File.write!(@yaml_path, emit())
    Mix.shell().info("wrote #{@yaml_path}")
  end

  @doc """
  Emit the yaml string. Pure; no file I/O. Used by both `run/1` and the
  CI gate (`mix esr.check_command_docs`).
  """
  @spec emit() :: String.t()
  def emit do
    specs = collect_specs()
    {slash_specs, internal_specs} = Enum.split_with(specs, &(&1.slash != :none))

    payload = %{
      "schema_version" => 1,
      "slashes" =>
        slash_specs
        |> Enum.sort_by(& &1.kind)
        |> Map.new(fn s -> {s.slash, slash_entry(s)} end),
      "internal_kinds" =>
        internal_specs
        |> Enum.sort_by(& &1.kind)
        |> Map.new(fn s -> {Atom.to_string(s.kind), internal_entry(s)} end)
    }

    @banner <> Ymlr.document!(payload, sort_maps: true)
  end

  defp collect_specs do
    [
      "Elixir.Esr.Commands.",
      "Elixir.Esr.Plugins."
    ]
    |> Enum.flat_map(&modules_with_prefix/1)
    |> Enum.filter(&function_exported?(&1, :command_meta, 0))
    |> Enum.map(& &1.command_meta())
    |> Enum.uniq_by(& &1.kind)
  end

  defp modules_with_prefix(prefix) do
    {:ok, mods} = :application.get_key(:esr, :modules)

    Enum.filter(mods, fn m ->
      String.starts_with?(Atom.to_string(m), prefix)
    end)
  end

  defp slash_entry(%Spec{} = s) do
    %{
      "kind" => Atom.to_string(s.kind),
      "permission" => s.permission,
      "command_module" => mod_string_for(s.kind),
      "requires_workspace_binding" => s.requires_workspace_binding,
      "requires_user_binding" => s.requires_user_binding,
      "category" => s.category,
      "description" => s.description,
      "args" => Enum.map(s.args, &arg_entry/1)
    }
  end

  defp internal_entry(%Spec{} = s) do
    %{
      "permission" => s.permission,
      "command_module" => mod_string_for(s.kind)
    }
  end

  defp arg_entry(%Arg{name: n, required: req, default: d}) do
    base = %{"name" => Atom.to_string(n), "required" => req}
    if d != nil, do: Map.put(base, "default", d), else: base
  end

  # Reverse-lookup: given a kind atom, find the module that emitted it.
  defp mod_string_for(kind) do
    {:ok, mods} = :application.get_key(:esr, :modules)

    mods
    |> Enum.find(fn m ->
      function_exported?(m, :command_meta, 0) and m.command_meta().kind == kind
    end)
    |> Atom.to_string()
    |> String.replace_prefix("Elixir.", "")
  end
end
```

```elixir
defmodule Mix.Tasks.Esr.GenCommandDocs do
  @moduledoc """
  Regenerate `docs/grammar/commands.md` and `docs/grammar/errors.md`
  from every command module's `command_meta/0`.

  Usage: `mix esr.gen_command_docs`
  """

  use Mix.Task
  alias Esr.Commands.Meta.Spec

  @commands_path "../docs/grammar/commands.md"
  @errors_path "../docs/grammar/errors.md"

  @banner """
  <!-- AUTOGENERATED — DO NOT EDIT BY HAND.
       Source: command_meta/0 in each Esr.Commands.* module.
       Regenerate: mix esr.gen_command_docs
       Drift in this file fails CI via mix esr.check_command_docs. -->
  """

  @shortdoc "Emit docs/grammar/{commands,errors}.md from command_meta/0"

  def run(_args) do
    Mix.Task.run("compile")
    Application.ensure_all_started(:esr)

    File.mkdir_p!(Path.dirname(@commands_path))
    File.write!(@commands_path, emit_commands_md())
    File.write!(@errors_path, emit_errors_md())
    Mix.shell().info("wrote #{@commands_path}, #{@errors_path}")
  end

  @spec emit_commands_md() :: String.t()
  def emit_commands_md do
    specs = collect_specs() |> Enum.reject(&(&1.slash == :none))

    sections =
      specs
      |> Enum.group_by(& &1.category)
      |> Enum.sort_by(fn {cat, _} -> cat end)
      |> Enum.map_join("\n\n", fn {cat, ss} ->
        body = ss |> Enum.sort_by(& &1.kind) |> Enum.map_join("\n", &render_command_entry/1)
        "## #{cat}\n\n#{body}"
      end)

    @banner <> "\n# ESR Command Reference\n\n" <> sections <> "\n"
  end

  @spec emit_errors_md() :: String.t()
  def emit_errors_md do
    specs = collect_specs()

    rows =
      specs
      |> Enum.flat_map(fn s ->
        Enum.map(s.errors, fn e ->
          {e.code, s.kind, e.message || "(no message)"}
        end)
      end)
      |> Enum.sort_by(fn {code, kind, _} -> {code, kind} end)
      |> Enum.map_join("\n", fn {code, kind, msg} ->
        "| `#{code}` | `#{kind}` | #{msg} |"
      end)

    @banner <>
      """

      # ESR Error Reference

      | type | kind | message |
      | ---- | ---- | ------- |
      """ <> rows <> "\n"
  end

  defp render_command_entry(%Spec{} = s) do
    args =
      Enum.map_join(s.args, " ", fn a ->
        if a.required, do: "#{a.name}=<…>", else: "[#{a.name}=<…>]"
      end)

    arg_seg = if args == "", do: "", else: " " <> args

    """
    ### `#{s.slash}`#{arg_seg}

    #{s.description}

    - **Kind:** `#{s.kind}`
    - **Permission:** `#{s.permission || "(none)"}`
    - **Requires user binding:** #{s.requires_user_binding}
    - **Requires workspace binding:** #{s.requires_workspace_binding}
    - **Errors:** #{Enum.map(s.errors, & "`" <> Atom.to_string(&1.code) <> "`") |> Enum.join(", ")}
    """
  end

  defp collect_specs do
    {:ok, mods} = :application.get_key(:esr, :modules)

    mods
    |> Enum.filter(&function_exported?(&1, :command_meta, 0))
    |> Enum.map(& &1.command_meta())
    |> Enum.uniq_by(& &1.kind)
  end
end
```

```elixir
defmodule Mix.Tasks.Esr.CheckCommandDocs do
  @moduledoc """
  CI gate: regenerates slash-routes.default.yaml + docs/grammar/{commands,errors}.md
  in memory and diffs against on-disk copies. Banners are stripped
  before diffing (so banner edits don't masquerade as content changes),
  but a MISSING banner is itself a failure.

  Usage: `mix esr.check_command_docs`. Exits 0 on success, 1 on drift.
  """

  use Mix.Task

  @yaml_path "priv/slash-routes.default.yaml"
  @commands_path "../docs/grammar/commands.md"
  @errors_path "../docs/grammar/errors.md"

  @shortdoc "Verify slash-routes.yaml + docs/grammar/* match command_meta/0"

  def run(_args) do
    Mix.Task.run("compile")
    Application.ensure_all_started(:esr)

    case check() do
      :ok ->
        Mix.shell().info("✅ slash-routes.yaml + docs/grammar/* in sync with command_meta/0")
        :ok

      {:error, file, reason} ->
        Mix.shell().error("""
        ❌ #{file} drift detected: #{reason}

        Regenerate with:
          mix esr.gen_slash_routes
          mix esr.gen_command_docs

        Then commit the regenerated files.
        """)

        exit({:shutdown, 1})
    end
  end

  @doc "Returns :ok or {:error, file, reason}."
  def check do
    with :ok <- check_file(@yaml_path, Mix.Tasks.Esr.GenSlashRoutes.emit()),
         :ok <- check_file(@commands_path, Mix.Tasks.Esr.GenCommandDocs.emit_commands_md()),
         :ok <- check_file(@errors_path, Mix.Tasks.Esr.GenCommandDocs.emit_errors_md()) do
      :ok
    end
  end

  defp check_file(path, generated) do
    case File.read(path) do
      {:ok, on_disk} ->
        verify_banner!(path, on_disk)

        if strip_banner(on_disk) == strip_banner(generated) do
          :ok
        else
          {:error, path, "content drift (run gen task to regenerate)"}
        end

      {:error, _} ->
        {:error, path, "file not found"}
    end
  end

  @doc false
  def strip_banner(content) do
    content
    |> String.replace(~r/^# AUTOGENERATED.*?\n(?:#.*?\n)+/s, "")
    |> String.replace(~r/^<!-- AUTOGENERATED.*?-->\n/s, "")
  end

  @doc false
  def verify_banner!(path, content) do
    if content =~ "AUTOGENERATED" do
      :ok
    else
      raise "missing AUTOGENERATED banner in #{path}; refuse to gate without it"
    end
  end
end
```

In `runtime/lib/esr/resource/slash_route/registry.ex`, add a new public function near line 100 (after `lookup/1`):

```elixir
  @doc """
  Fall-back resolution for input that `lookup/1` returned :not_found
  for. Returns one of:

    * `{:resource_help, nil}` — input was just `/`; render top-level resources
    * `{:resource_help, resource}` — input was `/workspace` or `/workspace:help`
    * `{:unknown_method, resource, method}` — input was `/workspace:bogus`
      where `workspace` is a known resource but `bogus` isn't a method
    * `{:unknown_resource, name}` — first segment doesn't match any resource
    * `:no_match` — input is a real slash (caller should have used lookup/1)

  Resources are derived from the set of `/<resource>:*` slash entries
  in the registry.
  """
  @spec lookup_prefix(String.t()) :: {:resource_help, String.t() | nil} | {:unknown_method, String.t(), String.t()} | {:unknown_resource, String.t()} | :no_match
  def lookup_prefix("/"), do: {:resource_help, nil}

  def lookup_prefix("/" <> rest) when is_binary(rest) do
    case String.split(rest, ":", parts: 2) do
      [resource, "help"] ->
        if known_resource?(resource), do: {:resource_help, resource}, else: {:unknown_resource, resource}

      [resource, method] ->
        cond do
          slash_exists?("/" <> resource <> ":" <> method) -> :no_match
          known_resource?(resource) -> {:unknown_method, resource, method}
          true -> {:unknown_resource, resource}
        end

      [resource] ->
        cond do
          slash_exists?("/" <> resource) -> :no_match
          known_resource?(resource) -> {:resource_help, resource}
          true -> {:unknown_resource, resource}
        end
    end
  end

  def lookup_prefix(_), do: :no_match

  defp known_resource?(name) do
    list_slashes()
    |> Enum.any?(fn route ->
      slash = route[:slash] || ""
      String.starts_with?(slash, "/" <> name <> ":")
    end)
  end

  defp slash_exists?(slash) do
    list_slashes()
    |> Enum.any?(&(&1[:slash] == slash))
  end
```
